SELECT ?person ?personLabel ?viwikiURL ?genderLabel ?countryLabel ?dob
WHERE {
  VALUES ?country { wd:Q30 }        # Quốc gia: Mỹ
  VALUES ?gender { wd:Q6581072 }    # Giới tính: Nữ wd:Q6581072
  VALUES ?occupation { wd:Q33999 wd:Q2526255 } # Diễn viên, Đạo diễn

  ?person wdt:P27 ?country;
          wdt:P21 ?gender;
          wdt:P106 ?occupation;
          wdt:P569 ?dob.

  # Bộ lọc tuổi (năm sinh)
  FILTER(?dob >= "1980-01-01T00:00:00Z"^^xsd:dateTime &&
         ?dob <= "2000-12-31T23:59:59Z"^^xsd:dateTime)

  # Bài Wikipedia tiếng Việt
  ?article schema:about ?person;
           schema:isPartOf <https://vi.wikipedia.org/>.
  BIND(STR(?article) AS ?viwikiURL)

  SERVICE wikibase:label { bd:serviceParam wikibase:language "en". }
}
LIMIT 1000
